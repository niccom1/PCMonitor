cmake_minimum_required(VERSION 3.20)
project(PCMonitor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_definitions(-D_WIN32_WINNT=0x0601 -DNTDDI_VERSION=0x06010000 -DWINAPI_FAMILY=WINAPI_FAMILY_DESKTOP_APP -DWIN32_LEAN_AND_MEAN -DNOMINMAX -DUNICODE -D_UNICODE)
add_executable(PCMonitor src/config.cpp src/util.cpp src/metrics.cpp src/collector.cpp src/tray.cpp src/main.cpp)
target_include_directories(PCMonitor PRIVATE ${CMAKE_SOURCE_DIR}/third_party ${CMAKE_SOURCE_DIR}/src)
target_link_libraries(PCMonitor PRIVATE ws2_32 pdh iphlpapi ole32 oleaut32 wbemuuid dxgi advapi32 shell32 user32 gdi32)
if(MINGW)
 target_link_options(PCMonitor PRIVATE -static -static-libgcc -static-libstdc++)
endif()
